2024-01-01 open Assets:Cash USD
2024-01-01 open Assets:Stock HOOL "FIFO"
2024-01-01 open Equity:Opening USD

  Assets:Cash 1000.00 USD
  Equity:Opening -1000.00 USD
'''
trade='2024-02-01 * "Purchase"\n'
trade+=('  bank_id: "B1"\n' if mode=='native-id' else '  import-id: "'+legacy+'"\n')
trade+='  Assets:Stock 3 HOOL {'+cost+', 2024-02-01, "'+label+'"}\n  Assets:Cash -'+str(cash)+'.00 USD\n'
ledger.write_text(base if mode=='current-id' else base+trade)
source.write_text('2024-02-01 * "Purchase"\n'+('  bank_id: "B1"\n' if mode=='native-id' else '')+
                  '  Assets:Stock 3 HOOL {'+cost+', 2024-02-01, "'+('Other' if mode=='different-label' else 'Café')+'"}\n  Assets:Cash -'+str(cash)+'.00 USD\n')
config.write_text('''import unicodedata
from beancount.parser import parser
from beancount.core.data import Transaction
class Reader:
    name='trade'
    def identify(self,file):return str(file).endswith('.statement')
    def account(self,file):return 'Assets:Stock'
    def extract(self,file,existing=None):
        entries,errors,_=parser.parse_file(str(file));assert not errors,errors
        return [e._replace(postings=[p._replace(cost=p.cost._replace(label=unicodedata.normalize(FORM,p.cost.label)))
                if p.cost is not None else p for p in e.postings]) if isinstance(e,Transaction) else e for e in entries]
CONFIG=[Reader()]
'''.replace('FORM',repr(form)))

args=['import',source,'--config',config]
if mode=='current-id':run('seed-current',[*args,'--apply'])
(case/'before-current.bean').write_bytes(ledger.read_bytes())
run('baseline',['check'])
run('preview',args)
run('default',[*args,'--apply'],human)
run('skip',[*args,'--duplicates','skip','--apply'])
run('facts',['query',"SELECT count(*),sum(number) WHERE account='Assets:Stock'"])
run('cash',['query',"SELECT sum(number) WHERE account='Assets:Cash'"])
run('check',['check'])
