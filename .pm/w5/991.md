A calculation plugin doubles a 10 USD transaction and divides its calculated
expense into two postings. That three-posting copy keeps the source header,
date, location and bank reference. The written transaction has two postings.
Both loaded rows are wrongly called written, `--on-disk` keeps both, and a
20 USD correction under the original bank ID is falsely skipped as an exact
duplicate at exit 0. The file declares only 10 USD. Expected: classify the
calculated copy as generated and refuse the changed bank amount as a conflict,
exit 4, with unchanged files. Do not automatically write the correction.

Replace the absolute directory below with a fresh disposable directory, run,
then repeat in another pristine directory. Other modes are controls and
adjacent states. The inspected local plugin performs only deterministic
in-memory arithmetic, with no I/O or services; it is executed through the
real native loader, not a mocked response. Children have isolated config/cache,
closed stdin/no TTY, bounded timeouts, separate streams and source hashes.
HOME and the global CLI installation stay unchanged.

```python
import hashlib, json, os, pathlib, subprocess
case = pathlib.Path('/abs/split-copy-case')
mode = 'split'  # same-count, changed-header, reverse, inferred
case.mkdir(parents=True)
plugin = '''from decimal import Decimal
from beancount.core.data import Transaction
__plugins__ = ("derive",)
def derive(entries, options, config=None):
    out = []
    for entry in entries:
        if not isinstance(entry, Transaction) or entry.narration != "Source row":
            out.append(entry)
            continue
        posts = [p._replace(units=p.units._replace(number=p.units.number*Decimal(2)))
                 for p in entry.postings]
        if SPLIT:
            second = posts[1]
            half = second._replace(units=second.units._replace(number=second.units.number/Decimal(2)))
            posts[1:] = [half,half]
        copy = entry._replace(meta=entry.meta.copy(),postings=posts,
                              narration="Derived row" if HEADER else entry.narration)
        out.extend([copy,entry] if REVERSE else [entry,copy])
    return out, []
'''
plugin = plugin.replace('SPLIT',str(mode!='same-count')).replace('HEADER',str(mode=='changed-header')).replace('REVERSE',str(mode=='reverse'))
module = case/'qa_derive.py'
module.write_text(plugin,encoding='utf-8')
ledger = case/'main.bean'
ledger.write_text('option "operating_currency" "USD"\n'
                  'option "insert_pythonpath" "TRUE"\nplugin "qa_derive"\n'
                  '2024-01-01 open Assets:Cash USD\n'
                  '2024-01-01 open Expenses:Food USD\n'
                  '2024-02-01 * "Source row"\n  bank_id: "source-10"\n'
                  '  Assets:Cash -10.00 USD\n'
                  + ('  Expenses:Food\n' if mode=='inferred' else '  Expenses:Food 10.00 USD\n'),
                  encoding='utf-8')
source = case/'correction.csv'
source.write_text('Date,Description,Amount,Reference\n2024-02-01,Source row,-20.00,source-10\n')
original = {p:p.read_bytes() for p in (ledger,module)}
env = os.environ.copy()
for key in ('BEA_FILE','BEA_TOKEN','BEANCOUNT_LOAD_CACHE_FILENAME'):
    env.pop(key,None)
env.update(BEA_CONFIG_DIR=str(case/'config'),XDG_CACHE_HOME=str(case/'cache'),
           BEA_NO_UPDATE_NOTIFIER='1')
def run(tag,args):
    cmd = ['bea','--json','--no-input','-f',str(ledger),*map(str,args)]
    r = subprocess.run(cmd,cwd=case,env=env,stdin=subprocess.DEVNULL,
                       capture_output=True,text=True,timeout=25)
    print(json.dumps(dict(tag=tag,args=cmd,exit=r.returncode,stdout=r.stdout,stderr=r.stderr,
                          files_preserved=all(p.read_bytes()==v for p,v in original.items()),
                          ledger_sha256=hashlib.sha256(ledger.read_bytes()).hexdigest())),flush=True)
run('check',['check'])
run('full',['list','transaction','--sort','oldest','--limit','100'])
run('on-disk',['list','transaction','--on-disk','--sort','oldest','--limit','100'])
run('limited',['list','transaction','--on-disk','--sort','newest','--limit','1'])
args = ['import',source,'--csv','date=Date,narration=Description,amount=Amount,id=Reference',
        '--account','Assets:Cash','--default-account','Expenses:Food']
run('preview-20',args)
run('apply-20',[*args,'--apply'])
run('facts',['query','SELECT account,sum(number) GROUP BY account ORDER BY account'])
source.write_text(source.read_text().replace('-20.00','-10.00'))
run('repeat-10',[*args,'--apply'])
```
