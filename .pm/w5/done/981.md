# Preserve acquisition labels in import identities

**Major · cli · 45–60m · A1.** Two separate positive acquisitions that differ
only by an absent versus explicitly empty cost label get the same generated
import ID. The second silently skips even with `--duplicates include`, exit
0, leaving **one share / Cash 900 USD instead of two shares / Cash 800 USD**.
The valid reference has separate `label: null` and `label: ""` positions.
No product fix implemented.

`4de7310eb`), checkout-selected `bea 0.3.1`, macOS 26.5.1 arm64, frontend
Python 3.13.3 / existing managed Python 3.12.10, Beancount 3.2.3. Beangulp is
already available. Fresh synthetic books/processes, closed stdin/no TTY,
isolated child config/cache, notifier disabled, no inherited ledger/token or

With `bea` on PATH, replace `/abs/empty-lot-case` below with a fresh absolute
disposable directory and run this standalone Python recipe. The inspected
local importer simply parses the two source statements. All purchases have
explicit positive units, per-unit cost and acquisition date. There is no

```python
from pathlib import Path
import json, os, subprocess

case = Path('/abs/empty-lot-case')
reverse = False
named_control = False
distinct_native_ids = False
native_change = False
case.mkdir(parents=True, exist_ok=False)
ledger = case / 'main.bean'
fresh = case / 'reference.bean'
base = '''option "operating_currency" "USD"
2024-01-01 open Assets:Stock HOOL "FIFO"
2024-01-01 open Assets:Cash USD
2024-01-01 open Equity:Opening USD
2024-01-01 * "Capital"
  Assets:Cash 1000 USD
  Equity:Opening -1000 USD
'''
ledger.write_text(base)
texts = []
for index in (0, 1):
    suffix = '' if index == 0 else ', ""'
    if reverse:
        suffix = ', ""' if index == 0 else ''
    if named_control and index == 1:
        suffix = ', "lot-b"'
    if native_change and index == 0:
        suffix = ', "lot-a"'
    bank = ''
    if distinct_native_ids or native_change:
        bank = f'  bank_id: "B{0 if native_change else index}"\n'
    texts.append('2024-02-01 * "Purchase"\n' + bank +
                 f'  Assets:Stock 1 HOOL {{100 USD, 2024-02-01{suffix}}}\n'
                 '  Assets:Cash -100 USD\n')
first, second = case / 'first.statement', case / 'second.statement'
first.write_text(texts[0])
second.write_text(texts[1])
# This independently valid ledger shows the purchases the inputs describe.
# Under a reused native ID it instead shows the changed acquisition alone.
fresh.write_text(base + (texts[1] if native_change else ''.join(texts)))
config = case / 'importer.py'
config.write_text('''from beancount.parser import parser
class Broker:
    name = 'broker'
    def identify(self, file): return str(file).endswith('.statement')
    def account(self, file): return 'Assets:Stock'
    def extract(self, file, existing_entries=None):
        entries, errors, _ = parser.parse_file(str(file))
        assert not errors, errors
        return entries
CONFIG = [Broker()]
''')
inputs = {p:p.read_bytes() for p in (first, second, config)}
env = os.environ.copy()
for key in ('BEA_FILE', 'BEA_TOKEN', 'BEANCOUNT_LOAD_CACHE_FILENAME',
            'BEA_ENGINE_PYTHON', 'BEA_ENGINE_DIR'):
    env.pop(key, None)
env.update(BEA_CONFIG_DIR=str(case / 'config'), XDG_CACHE_HOME=str(case / 'cache'),
           BEA_NO_UPDATE_NOTIFIER='1')
def run(tag, args, file=ledger, json_mode=True):
    argv = ['bea', *(['--json'] if json_mode else []), '--no-input', '-f', str(file), *map(str, args)]
    before = file.read_bytes()
    result = subprocess.run(argv, cwd=case, env=env, stdin=subprocess.DEVNULL,
                            capture_output=True, text=True, timeout=25)
    assert all(p.read_bytes() == data for p, data in inputs.items())
    print(json.dumps(dict(tag=tag, args=argv, exit=result.returncode,
                         stdout=result.stdout, stderr=result.stderr,
                         ledger_changed=file.read_bytes() != before)), flush=True)
def importing(file):
    return ['import', file, '--config', config]
facts = ['query', 'SELECT account,sum(number) GROUP BY account ORDER BY account']
lots = ['query', "SELECT sum(position) WHERE account='Assets:Stock'"]
run('baseline', ['check'])
run('first', [*importing(first), '--apply'])
run('unchanged-first', [*importing(first), '--apply'])
run('second-preview', [*importing(second), '--duplicates', 'include'])
run('second-include', [*importing(second), '--duplicates', 'include', '--apply'])
run('facts', facts)
run('lots', lots)
run('check', ['check'])
run('reference-facts', facts, file=fresh)
run('reference-lots', lots, file=fresh)
run('reference-check', ['check'], file=fresh)
```

**Acquisition label normalization, 2026-10-22.** The same pending label
projection/hash repair also needs to compare canonically equivalent labels
as the CLI loader reads them. A Python broker importer that builds upstream
`Transaction`/`CostSpec` objects from CSV keeps a decomposed `Café` label.
The first write succeeds; reloading reads that authored label in NFC, but
incoming identity still compares its original NFD bytes. On HEAD
`7dcf21e19` this reproduced twice, and the standalone recipe below is
verified on `75cff6325` after pulling the unrelated short-cover and parser
interpolation repairs. Only acquisitions with explicit units, cost, date
and ordinary nonempty labels are involved. This is a Python importer reading
a CSV, rather than the built-in `--csv` mapper, which has no lot fields.

- With ASCII native ID B1, importing the identical NFD file again previews
  `conflict` and apply exits **4**, even under include, preserving one share
  and Cash **900 USD**. Its advice says the stable ID has different data,
  although no input bytes changed. NFC throughout, or changing only the
  second label to NFC, correctly returns duplicate / written 0.
- Without a native ID, first importing NFC then the equivalent NFD label
  previews `new` and the default apply writes a second purchase, exit **0**:
  **two shares / Cash 800 USD**, instead of one share / Cash 900. A third
  identical NFD import skips; the two different generated IDs now coexist.
- An existing hand-entered NFC purchase is likewise classified `new` by an
  NFD incoming label. `--duplicates skip --apply` still writes the duplicate,
  exit 0. Incoming NFC instead recognizes a possible hand-entered twin and
  skip writes 0. Fresh checks pass in both books.
- NFD followed by NFC under generated IDs becomes `possible_duplicate`,
  rather than exact duplicate; default apply exits 4 and include writes a
  second purchase. Identical NFD generated-ID imports already skip and must
  continue doing so. A genuinely different named acquisition under the same

  genuinely new.

`mode = 'native-nfd'` shows the unchanged-export conflict; use `hash-reverse`

```python
from pathlib import Path
import csv, json, os, subprocess, unicodedata

case = Path('/abs/unicode-lot-case')
mode = 'native-nfd'
human_second = False
# native-nfd: identical file conflicts; hash-reverse: equivalent label writes twice.
# manual-nfd: --duplicates skip still appends the existing purchase.
profiles = {
    'native-nfd': (True, 'NFD', 'NFD', False, False),
    'native-nfc': (True, 'NFC', 'NFC', False, False),
    'native-switch': (True, 'NFD', 'NFC', False, False),
    'native-different': (True, 'NFC', 'NFC', False, True),
    'hash-nfd': (False, 'NFD', 'NFD', False, False),
    'hash-switch': (False, 'NFD', 'NFC', False, False),
    'hash-reverse': (False, 'NFC', 'NFD', False, False),
    'manual-nfd': (False, 'NFC', 'NFD', True, False),
    'manual-nfc': (False, 'NFC', 'NFC', True, False),
    'manual-different': (False, 'NFC', 'NFC', True, True),
}
native_id, first_form, second_form, manual, different = profiles[mode]
case.mkdir(parents=True, exist_ok=False)
ledger, source, config = case/'main.bean', case/'broker.csv', case/'importer.py'
base = '''option "operating_currency" "USD"
2024-01-01 open Assets:Cash USD
2024-01-01 open Assets:Stock HOOL "FIFO"
2024-01-01 open Equity:Opening USD
2024-01-01 * "Capital"
  Assets:Cash 1000.00 USD
  Equity:Opening -1000.00 USD
'''
seed = '''2024-02-01 * "Purchase"
  Assets:Stock 1 HOOL {100 USD, 2024-02-01, "Café"}
  Assets:Cash -100.00 USD
'''
ledger.write_text(base + (seed if manual else ''), encoding='utf-8')
# An ordinary broker CSV importer, constructing explicit upstream objects.
config.write_text('''import csv, datetime
from decimal import Decimal as D
from beancount.core.amount import Amount
from beancount.core.data import Posting, Transaction, new_metadata
from beancount.core.position import CostSpec
class Broker:
    name = 'broker'
    def identify(self, file): return str(file).endswith('.csv')
    def account(self, file): return 'Assets:Stock'
    def extract(self, file, existing_entries=None):
        entries = []
        with open(file, encoding='utf-8', newline='') as stream:
            for row in csv.DictReader(stream):
                day = datetime.date.fromisoformat(row['Date'])
                meta = new_metadata(file, 2)
                if row['ID']: meta['bank_id'] = row['ID']
                cost = CostSpec(D('100'), None, 'USD', day, row['Lot'], False)
                entries.append(Transaction(meta, day, '*', None, 'Purchase',
                    frozenset(), frozenset(), [
                        Posting('Assets:Stock', Amount(D('1'), 'HOOL'), cost, None, None, None),
                        Posting('Assets:Cash', Amount(D('-100.00'), 'USD'), None, None, None, None)]))
        return entries
CONFIG = [Broker()]
''', encoding='utf-8')
def write_source(form, changed=False):
    with source.open('w', encoding='utf-8', newline='') as stream:
        writer = csv.writer(stream)
        writer.writerow(['Date', 'ID', 'Lot'])
        writer.writerow(['2024-02-01', 'B1' if native_id else '',
                         'Other' if changed else unicodedata.normalize(form, 'Café')])
env = os.environ.copy()
for key in ('BEA_FILE', 'BEA_TOKEN', 'BEANCOUNT_LOAD_CACHE_FILENAME',
            'BEA_ENGINE_PYTHON', 'BEA_ENGINE_DIR'):
    env.pop(key, None)
env.update(BEA_CONFIG_DIR=str(case/'config'), XDG_CACHE_HOME=str(case/'cache'),
           BEA_NO_UPDATE_NOTIFIER='1')
def run(tag, args, human=False):
    argv = ['bea', *([] if human else ['--json']), '--no-input', '-f', str(ledger), *map(str, args)]
    before, csv_before, config_before = ledger.read_bytes(), source.read_bytes(), config.read_bytes()
    result = subprocess.run(argv, cwd=case, env=env, stdin=subprocess.DEVNULL,
                            capture_output=True, text=True, timeout=25)
    assert source.read_bytes() == csv_before and config.read_bytes() == config_before
    print(json.dumps(dict(tag=tag, args=argv, exit=result.returncode,
                         stdout=result.stdout, stderr=result.stderr,
                         ledger_changed=ledger.read_bytes()!=before)), flush=True)
args = ['import', source, '--config', config]
write_source(first_form)
run('baseline', ['check'])
if not manual: run('first-apply', [*args, '--apply'])
write_source(second_form, different)
run('second-preview', args)
run('second-default', [*args, *(['--duplicates', 'skip'] if manual else []), '--apply'], human_second)
if not manual: run('second-include', [*args, '--duplicates', 'include', '--apply'])
run('facts', ['query', 'SELECT account,sum(number) GROUP BY account ORDER BY account'])
run('positions', ['query', "SELECT sum(position) WHERE account='Assets:Stock'"])
run('final-check', ['check'])
```
