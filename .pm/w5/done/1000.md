# Match authored notes, events and customs during imports

**Minor · cli · 35–50m · A1.** Import's non-transaction duplicate index uses
the loaded plugin view. A filter or singleton transformation therefore makes
an identical authored note, event and custom directive append repeatedly,
even with explicit skip. A change matching only the transformed view falsely
skips instead. These are nonfinancial directive duplicates; cash remains
1000 USD. No product fix implemented.

Real local CLI QA 2026-10-23 at
`53b04d9f8e4e53ec9e1b96145add09ff9878acfd`, `bea 0.3.1`, macOS 26.5.1 arm64,
frontend Python 3.13.3, managed Python 3.12.10 / Beancount 3.2.3. The checkout
supplies frontend/helper source. Fresh bounded subprocesses use closed
stdin/no TTY, separate configuration/cache and notifier-off. The inspected
importer only parses its local statement; the inspected plugin only filters
or replaces three ordinary directive values. No service, credential, real
ledger or dependency change is involved.

With `bea` on PATH, select a fresh absolute scratch directory and run the
complete recipe below. It first imports all three directives through the real
CLI, then repeats under default policy and explicit skip. Plugin-free snapshots
remove only this fixture's two settings lines, so listing them exposes what
was actually authored without changing the live plugin view.

```python
from pathlib import Path
import hashlib, json, os, subprocess, time

case = Path('/abs/filtered-directives-case')
view = 'filter'  # plain, pass, transform
changed = False
human_repeat = False
case.mkdir(parents=True, exist_ok=False)
ledger, source, config = case/'main.bean', case/'entries.statement', case/'importer.py'
module = case/'qa_view.py'
module.write_text("from beancount.core.data import Event,Note,Custom\n__plugins__=('apply_view',)\ndef apply_view(entries,options_map,mode):\n    result=[]\n    for entry in entries:\n        if isinstance(entry,(Event,Note,Custom)):\n            if mode=='filter':continue\n            if mode=='transform':\n                if isinstance(entry,Event):entry=entry._replace(description='Rome')\n                elif isinstance(entry,Note):entry=entry._replace(comment='Pending')\n                else:entry=entry._replace(values=[entry.values[0]._replace(value='Pending')])\n        result.append(entry)\n    return result,[]\n")
settings = '' if view == 'plain' else 'option "insert_pythonpath" "TRUE"\nplugin "qa_view" "'+view+'"\n'
base = '2024-01-01 open Assets:Cash USD\n2024-01-01 open Equity:Opening USD\n2024-01-01 * "Capital"\n  Assets:Cash 1000.00 USD\n  Equity:Opening -1000.00 USD\n'
ledger.write_text(settings+base)
def statement(changed=False):
    return ('2024-02-01 event "location" "'+('Rome' if changed else 'Paris')+'"\n'
            +'2024-02-01 note Assets:Cash "'+('Pending' if changed else 'Reviewed')+'"\n'
            +'2024-02-01 custom "review-note" "'+('Pending' if changed else 'Paid')+'"\n')
source.write_text(statement())
config.write_text("from beancount.parser import parser\nclass Reader:\n    name='local-directives'\n    def identify(self,file):return str(file).endswith('.statement')\n    def account(self,file):return 'Assets:Cash'\n    def extract(self,file,existing_entries=None):\n        entries,errors,_=parser.parse_file(str(file))\n        assert not errors,errors\n        return entries\nCONFIG=[Reader()]\n")
env = os.environ.copy()
for key in ('BEA_FILE','BEA_TOKEN','BEANCOUNT_LOAD_CACHE_FILENAME','BEA_ENGINE_PYTHON','BEA_ENGINE_DIR'):
    env.pop(key,None)
overrides=dict(BEA_CONFIG_DIR=str(case/'config'),XDG_CACHE_HOME=str(case/'cache'),BEA_NO_UPDATE_NOTIFIER='1')
env.update(overrides)
def run(tag,args,target=ledger,human=False):
    argv=['bea',*([] if human else ['--json']),'--no-input','-f',str(target),*map(str,args)]
    before=ledger.read_bytes();fixed={p:p.read_bytes() for p in (source,config,module)};start=time.monotonic()
    result=subprocess.run(argv,cwd=case,env=env,stdin=subprocess.DEVNULL,capture_output=True,text=True,timeout=25)
    assert all(p.read_bytes()==b for p,b in fixed.items())
    print(json.dumps(dict(tag=tag,args=argv,cwd=str(case),env=overrides,stdin_mode='DEVNULL',tty=False,
                         exit=result.returncode,stdout=result.stdout,stderr=result.stderr,duration=time.monotonic()-start,
                         ledger_changed=ledger.read_bytes()!=before,inputs_preserved=True,
                         input_sha256={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in fixed})),flush=True)
def snapshot(name):
    file=case/name
    text=ledger.read_text()
    assert text.startswith(settings)
    file.write_text(text[len(settings):])
    return file
args=['import',source,'--config',config]
run('baseline',['check']); run('first-preview',args); run('first-apply',[*args,'--apply'])
first=snapshot('first-written.bean')
if changed:source.write_text(statement(True))
run('repeat-preview',args);run('repeat-apply',[*args,'--apply'],human=human_repeat)
run('skip-repeat',[*args,'--duplicates','skip','--apply'])
last=snapshot('last-written.bean')
for label,file in [('first',first),('last',last)]:
    run(label+'-check',['check'],file)
    for kind in ('event','note','custom'):run(label+'-'+kind,['list',kind],file)
run('active-facts',['query','SELECT account,sum(number) GROUP BY account ORDER BY account'])
run('final-check',['check'])
```
