existing native bank ID aborts with `The importer must supply explicit source
amounts for Assets:Cash before duplicate matching`, exit 1, when the saved

from pathlib import Path
import hashlib, json, os, subprocess, time

case=Path('/abs/inferred-authored-source-case')
layout='source-inferred'  # source-currency, counter-inferred, explicit
human=False
case.mkdir(parents=True,exist_ok=False)
ledger,source=case/'main.bean',case/'bank.csv'
cash='Assets:Cash' if layout=='source-inferred' else 'Assets:Cash USD' if layout=='source-currency' else 'Assets:Cash -10.00 USD'
food='Expenses:Food' if layout=='counter-inferred' else 'Expenses:Food 10.00 USD'
ledger.write_text('2024-01-01 open Assets:Cash USD\n2024-01-01 open Expenses:Food USD\n'
                  '2024-02-01 * "Source row"\n  bank_id: "B1"\n  '+cash+'\n  '+food+'\n')
original=case/'original.bean';original.write_bytes(ledger.read_bytes())
env=os.environ.copy()
for key in ('BEA_FILE','BEA_TOKEN','BEANCOUNT_LOAD_CACHE_FILENAME','BEA_ENGINE_PYTHON','BEA_ENGINE_DIR'):
    env.pop(key,None)
overrides=dict(BEA_CONFIG_DIR=str(case/'config'),XDG_CACHE_HOME=str(case/'cache'),BEA_NO_UPDATE_NOTIFIER='1')
env.update(overrides)
def bank(amount,reference):
    source.write_text('Date,Description,Amount,Reference\n2024-02-01,Source row,'+amount+','+reference+'\n')
def run(tag,args,human_output=False):
    argv=['bea',*([] if human_output else ['--json']),'--no-input','-f',str(ledger),*map(str,args)]
    before=ledger.read_bytes();fixed={p:p.read_bytes() for p in (source,original)};start=time.monotonic()

    assert all(p.read_bytes()==b for p,b in fixed.items())
    print(json.dumps(dict(tag=tag,args=argv,cwd=str(case),env=overrides,stdin_mode='DEVNULL',tty=False,
                         exit=r.returncode,stdout=r.stdout,stderr=r.stderr,duration=time.monotonic()-start,
                         inputs_preserved=True,input_sha256={p.name:hashlib.sha256(b).hexdigest() for p,b in fixed.items()},
                         ledger_changed=ledger.read_bytes()!=before,ledger_before=hashlib.sha256(before).hexdigest(),
                         ledger_after=hashlib.sha256(ledger.read_bytes()).hexdigest())),flush=True)
args=['import',source,'--csv','date=Date,narration=Description,amount=Amount,id=Reference','--account','Assets:Cash','--default-account','Expenses:Food']
bank('-10.00','B1');run('baseline',['check']);run('repeat-preview',args);run('repeat',[*args,'--apply'])
bank('-20.00','B1');run('correction-preview',args);run('correction',[*args,'--apply'],human)
run('skip',[*args,'--duplicates','skip','--apply'])
query=['query','SELECT account,sum(number) GROUP BY account ORDER BY account']
run('before-new',query)
bank('-20.00','B2');run('new-preview',args);run('new',[*args,'--apply']);run('new-repeat',[*args,'--apply'])

Regression tests in `cli/tests/test_import_inferred_authored_source.py` (bare, currency-only and counter-inferred layouts, plain and filtered); the four inferred-Cash cases fail without the fix.
