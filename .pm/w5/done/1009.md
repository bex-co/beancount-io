```python
import hashlib,json,os,subprocess,time,unicodedata

mode='combined'  # nfc-compound, nfd-simple, current-id, native-id, different-label
human=False
case.mkdir(parents=True,exist_ok=False)
ledger,source,config=case/'main.bean',case/'purchase.statement',case/'importer.py'
form='NFC' if mode=='nfc-compound' else 'NFD'

cost='1 USD' if mode=='nfd-simple' else '-3 # 10 USD'
cash=3 if mode=='nfd-simple' else 1
legacy={'nfc-compound':'csv:sha256:0492875e1337d2e2',
        'nfd-simple':'csv:sha256:59ebc3b188e6287a'}.get(mode,'csv:sha256:345f500b8f6d41a0')
base='''option "operating_currency" "USD"
2024-01-01 open Assets:Cash USD
2024-01-01 open Assets:Stock HOOL "FIFO"
2024-01-01 open Equity:Opening USD
2024-01-01 * "Capital"
  Assets:Cash 1000.00 USD
  Equity:Opening -1000.00 USD
'''
trade='2024-02-01 * "Purchase"\n'
trade+=('  bank_id: "B1"\n' if mode=='native-id' else '  import-id: "'+legacy+'"\n')
trade+='  Assets:Stock 3 HOOL {'+cost+', 2024-02-01, "'+label+'"}\n  Assets:Cash -'+str(cash)+'.00 USD\n'
ledger.write_text(base if mode=='current-id' else base+trade)
source.write_text('2024-02-01 * "Purchase"\n'+('  bank_id: "B1"\n' if mode=='native-id' else '')+
                  '  Assets:Stock 3 HOOL {'+cost+', 2024-02-01, "'+('Other' if mode=='different-label' else 'Café')+'"}\n  Assets:Cash -'+str(cash)+'.00 USD\n')
config.write_text('''import unicodedata
from beancount.parser import parser
from beancount.core.data import Transaction
class Reader:
    name='trade'
    def identify(self,file):return str(file).endswith('.statement')
    def account(self,file):return 'Assets:Stock'
    def extract(self,file,existing=None):
        entries,errors,_=parser.parse_file(str(file));assert not errors,errors
        return [e._replace(postings=[p._replace(cost=p.cost._replace(label=unicodedata.normalize(FORM,p.cost.label)))
                if p.cost is not None else p for p in e.postings]) if isinstance(e,Transaction) else e for e in entries]
CONFIG=[Reader()]
'''.replace('FORM',repr(form)))
fixed={p:p.read_bytes() for p in (source,config)}
env=os.environ.copy()
for key in ('BEA_FILE','BEA_TOKEN','BEANCOUNT_LOAD_CACHE_FILENAME','BEA_ENGINE_PYTHON','BEA_ENGINE_DIR'):
    env.pop(key,None)
overrides=dict(BEA_CONFIG_DIR=str(case/'config'),XDG_CACHE_HOME=str(case/'cache'),BEA_NO_UPDATE_NOTIFIER='1')
env.update(overrides)
def run(tag,args,human_output=False):
    before=ledger.read_bytes();argv=['bea',*([] if human_output else ['--json']),'--no-input','-f',str(ledger),*map(str,args)]
    started=time.monotonic()
    r=subprocess.run(argv,cwd=case,env=env,stdin=subprocess.DEVNULL,capture_output=True,text=True,timeout=25)
    assert all(p.read_bytes()==b for p,b in fixed.items())
    print(json.dumps(dict(tag=tag,args=argv,cwd=str(case),stdin_mode='DEVNULL',tty=False,env=overrides,
                         exit=r.returncode,stdout=r.stdout,stderr=r.stderr,duration=time.monotonic()-started,
                         inputs_preserved=True,input_sha256={p.name:hashlib.sha256(b).hexdigest() for p,b in fixed.items()},
                         ledger_before=hashlib.sha256(before).hexdigest(),ledger_after=hashlib.sha256(ledger.read_bytes()).hexdigest(),
                         ledger_changed=ledger.read_bytes()!=before)),flush=True)
    if r.returncode:assert ledger.read_bytes()==before,'Inspect failed-write effects before retrying'
args=['import',source,'--config',config]
if mode=='current-id':run('seed-current',[*args,'--apply'])
(case/'before-current.bean').write_bytes(ledger.read_bytes())
run('baseline',['check'])
run('preview',args)
run('default',[*args,'--apply'],human)
run('skip',[*args,'--duplicates','skip','--apply'])
run('facts',['query',"SELECT count(*),sum(number) WHERE account='Assets:Stock'"])
run('cash',['query',"SELECT sum(number) WHERE account='Assets:Cash'"])
run('check',['check'])
```
