With `bea` on PATH and a fresh absolute directory, this complete first recipe
reproduces the netted case. The existing +1 at 150 USD / January 2 and −1 at
100 USD / January 3 share the label `First<LF>Lot`. A fully stated cover
should cancel the 100 USD short even though total labelled units net to zero.

```python
from pathlib import Path
import hashlib, json, os, subprocess, time

case = Path('/abs/netted-short-case')
mode = 'zero'
human_cover = False
long_units, short_units, long_label = {
    'zero': (1, 1, 'First\nLot'),
    'positive': (2, 1, 'First\nLot'),
    'negative': (1, 2, 'First\nLot'),
    'different': (1, 1, 'Long'),
    'short-only': (0, 1, 'First\nLot'),
}[mode]
label = 'First\nLot'
case.mkdir(parents=True, exist_ok=False)
ledger, reference = case/'main.bean', case/'reference.bean'
base = '''option "operating_currency" "USD"
2024-01-01 open Assets:Stock HOOL "NONE"
2024-01-01 open Assets:Cash USD
2024-01-01 open Income:Gains USD
2024-01-01 open Equity:Opening USD
2024-01-01 * "Capital"
  Assets:Cash 1000.00 USD
  Equity:Opening -1000.00 USD
'''
def trade(day, name, quantity, cost, acquired, lot, cash, gain=None):
    cost_data = dict(number=str(cost), currency='USD', date=acquired, label=lot)
    postings = [dict(account='Assets:Stock', units=dict(number=str(quantity),currency='HOOL'),cost=cost_data),
                dict(account='Assets:Cash', units=dict(number=str(cash),currency='USD'))]
    text = (f'{day} * "{name}"\n  Assets:Stock {quantity} HOOL {{{cost} USD, {acquired}, "{lot}"}}\n'
            f'  Assets:Cash {cash} USD\n')
    if gain is not None:
        postings.append(dict(account='Income:Gains', units=dict(number=str(gain),currency='USD')))
        text += f'  Income:Gains {gain} USD\n'
    return dict(date=day,narration=name,postings=postings), text
if long_units:
    base += trade('2024-01-02','Long lot',long_units,150,'2024-01-02',long_label,-150*long_units)[1]
base += trade('2024-01-03','Short lot',-short_units,100,'2024-01-03',label,100*short_units)[1]
cover, cover_text = trade('2024-02-01','Cover short',short_units,100,'2024-01-03',label,-90*short_units,-10*short_units)
drain, drain_text = trade('2024-03-01','Sell long at cost',-long_units,150,'2024-01-02',long_label,150*long_units)
ledger.write_text(base,encoding='utf-8')
reference.write_text(base+cover_text+(drain_text if long_units else ''),encoding='utf-8')
cover_file, drain_file = case/'cover.json', case/'drain.json'
cover_file.write_text(json.dumps([cover]),encoding='utf-8')
drain_file.write_text(json.dumps([drain]),encoding='utf-8')
inputs = {p:p.read_bytes() for p in (cover_file,drain_file)}
env = os.environ.copy()
for key in ('BEA_FILE','BEA_TOKEN','BEANCOUNT_LOAD_CACHE_FILENAME','BEA_ENGINE_PYTHON','BEA_ENGINE_DIR'):
    env.pop(key,None)
env.update(BEA_CONFIG_DIR=str(case/'config'),XDG_CACHE_HOME=str(case/'cache'),BEA_NO_UPDATE_NOTIFIER='1')
def run(tag,args,file=ledger,human=False):
    argv = ['bea',*([] if human else ['--json']),'--no-input','-f',str(file),*map(str,args)]
    before = file.read_bytes()
    started = time.monotonic()
    result = subprocess.run(argv,cwd=case,env=env,stdin=subprocess.DEVNULL,capture_output=True,text=True,timeout=25)
    assert all(p.read_bytes()==data for p,data in inputs.items())
    print(json.dumps(dict(tag=tag,args=argv,exit=result.returncode,stdout=result.stdout,stderr=result.stderr,
                         ledger_changed=file.read_bytes()!=before,cwd=str(case),stdin_mode='DEVNULL',tty=False,
                         duration=time.monotonic()-started,env={k:env[k] for k in ('BEA_CONFIG_DIR','XDG_CACHE_HOME','BEA_NO_UPDATE_NOTIFIER')},
                         input_sha256={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in inputs},
                         ledger_before=hashlib.sha256(before).hexdigest(),ledger_after=hashlib.sha256(file.read_bytes()).hexdigest())),flush=True)
positions = ['query',"SELECT sum(position) WHERE account='Assets:Stock'"]
closing = ['add','close','--account','Assets:Stock','--date','2024-03-02']
run('baseline',['check'])
run('reference-baseline',['check'],reference)
run('cover',['add','transactions','--from',cover_file],human=human_cover)
run('after-cover',positions)
if long_units: run('drain',['add','transactions','--from',drain_file])
run('after-drain',positions)
run('close',closing)
run('reference-close',closing,reference)
run('reference-positions',positions,reference)
run('facts',['query','SELECT account,sum(number) GROUP BY account ORDER BY account'])
run('final-check',['check'])
run('reference-check',['check'],reference)
```

```python
import hashlib, json, os, pathlib, subprocess, time
case = pathlib.Path('/abs/backdated-cover-case')
mode = 'future-flat'
# Other controls: future-positive, future-negative, plain, later-recorded.
case.mkdir(parents=True)
main = case / 'main.bean'
reference = case / 'reference.bean'
label = 'First Lot' if mode == 'plain' else 'First\nLot'
future_qty = 3 if mode == 'future-positive' else 1 if mode == 'future-negative' else 2
sale_qty = future_qty - 1
base = '''option "operating_currency" "USD"
2024-01-01 open Assets:Stock HOOL "NONE"
2024-01-01 open Assets:Cash USD
2024-01-01 open Income:Gains USD
2024-01-01 open Equity:Opening USD
2024-01-01 * "Capital"
  Assets:Cash 1000.00 USD
  Equity:Opening -1000.00 USD
''' + f'2024-01-02 * "Short lot"\n  Assets:Stock -2 HOOL {{100 USD, 2024-01-02, "{label}"}}\n  Assets:Cash 200.00 USD\n'
early = f'2024-02-01 * "Backfilled cover"\n  Assets:Stock 1 HOOL {{100 USD, 2024-01-02, "{label}"}}\n  Assets:Cash -90.00 USD\n  Income:Gains -10.00 USD\n'
future = f'2024-03-01 * "Later cover"\n  Assets:Stock {future_qty} HOOL {{100 USD, 2024-01-02, "{label}"}}\n  Assets:Cash {-100*future_qty}.00 USD\n'
sale = (f'2024-04-01 * "Sell resulting long"\n  Assets:Stock {-sale_qty} HOOL {{100 USD, 2024-01-02, "{label}"}}\n  Assets:Cash {100*sale_qty}.00 USD\n') if sale_qty else ''
main.write_text(base + ('' if mode == 'later-recorded' else future), encoding='utf-8')
(case / 'initial.bean').write_bytes(main.read_bytes())
reference.write_text(base + early + future + sale, encoding='utf-8')
def row(name, date, quantity, cash, gain=None):
    postings = [dict(account='Assets:Stock', units=dict(number=str(quantity), currency='HOOL'),
                     cost=dict(number='100', currency='USD', date='2024-01-02', label=label)),
                dict(account='Assets:Cash', units=dict(number=str(cash), currency='USD'))]
    if gain is not None:
        postings.append(dict(account='Income:Gains', units=dict(number=str(gain), currency='USD')))
    return dict(date=date, narration=name, postings=postings)
for name, value in [('early', row('Backfilled cover','2024-02-01',1,-90,-10)),
                    ('later', row('Later cover','2024-03-01',future_qty,-100*future_qty)),
                    ('sale', row('Sell resulting long','2024-04-01',-sale_qty,100*sale_qty))]:
    (case / (name + '.json')).write_text(json.dumps([value]), encoding='utf-8')
env = os.environ.copy()
for key in ('BEA_FILE', 'BEA_TOKEN', 'BEANCOUNT_LOAD_CACHE_FILENAME'):
    env.pop(key, None)
env.update(BEA_CONFIG_DIR=str(case / 'config'), XDG_CACHE_HOME=str(case / 'cache'),
           BEA_NO_UPDATE_NOTIFIER='1')
def run(tag, args, file=main):
    before = file.read_bytes()
    cmd = ['bea','--json','--no-input','--file',str(file),*map(str,args)]
    started = time.monotonic()
    r = subprocess.run(cmd, cwd=case, env=env, stdin=subprocess.DEVNULL,
                       capture_output=True, text=True, timeout=25)
    print(json.dumps(dict(tag=tag,args=cmd,exit=r.returncode,stdout=r.stdout,stderr=r.stderr,
                          file_preserved=file.read_bytes()==before,cwd=str(case),stdin_mode='DEVNULL',tty=False,
                          duration=time.monotonic()-started,env={k:env[k] for k in ('BEA_CONFIG_DIR','XDG_CACHE_HOME','BEA_NO_UPDATE_NOTIFIER')},
                          input_sha256={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in case.glob('*.json')},
                          ledger_before=hashlib.sha256(before).hexdigest(),ledger_after=hashlib.sha256(file.read_bytes()).hexdigest())), flush=True)
run('baseline', ['check'])
run('reference-baseline', ['check'], reference)
run('backfill', ['add','transactions','--from',case/'early.json'])
if mode == 'later-recorded':
    run('record-later', ['add','transactions','--from',case/'later.json'])
if sale_qty:
    run('sale', ['add','transactions','--from',case/'sale.json'])
run('positions', ['query', "SELECT sum(position) WHERE account='Assets:Stock'"])
run('facts', ['query', 'SELECT account,sum(number) GROUP BY account ORDER BY account'])
run('close', ['add','close','--date','2024-04-02','--account','Assets:Stock'])
run('check', ['check'])
run('reference-close', ['add','close','--date','2024-04-02','--account','Assets:Stock'], reference)
run('reference-check', ['check'], reference)
```
