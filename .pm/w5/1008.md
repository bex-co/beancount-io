# Reject conflicting imported balance assertions from authored facts

**Minor · cli · 35–50m · A1 · worker1.** Native import appends a different
same-day balance assertion when a plugin filters or rewrites assertions.
Preview says `new`, apply exits 0 / writes 1, and an exact repeat skips.
The active view checks valid, but the actual file now declares both 100.00 USD
and 100.01 ~ 0.02 USD for the same date/account/currency. A separate native
load of those declarations reports **Duplicate balance assertion with
different amounts**. Cash stays 100 USD; no changed cash or money loss is
claimed. No product fix implemented.

from pathlib import Path
import hashlib,json,os,subprocess,time

case=Path('/abs/import-balance-conflict-case')
view='filter'  # transform, pass
split=False
no_existing=False
new_date=False
human=False
skip=False
case.mkdir(parents=True,exist_ok=False)
module,importer,source=case/'qa_view.py',case/'importers.py',case/'statement.txt'
module.write_text('''from decimal import Decimal
from beancount.core.data import Balance
__plugins__=('view',)
def view(entries,options,mode):
    out=[]
    for e in entries:
        if isinstance(e,Balance):
            if mode=='filter':continue
            if mode=='transform':e=e._replace(amount=e.amount._replace(number=Decimal('100.01')),
                                             tolerance=Decimal('0.02'),meta=e.meta.copy())
        out.append(e)
    return out,[]
''')
importer.write_text('''from datetime import date
from decimal import Decimal
from beancount.core import data
from beancount.core.amount import Amount
class Statement:
    name='statement'
    def identify(self,filename):return filename.endswith('.txt')
    def account(self,filename):return 'Assets:Cash'
    def extract(self,filename,existing):
        with open(filename) as f:day,number,tolerance=f.read().strip().split(',')
        return [data.Balance(data.new_metadata(filename,1),date.fromisoformat(day),'Assets:Cash',
                             Amount(Decimal(number),'USD'),Decimal(tolerance),None)]
CONFIG=[Statement()]
''')
base='''2024-01-01 open Assets:Cash USD
2024-01-01 open Equity:Opening USD
2024-01-02 * "Capital"
  Assets:Cash 100.00 USD
  Equity:Opening -100.00 USD
'''
if not no_existing:base+='2024-02-01 balance Assets:Cash 100.00 USD\n'
settings='option "insert_pythonpath" "TRUE"\nplugin "qa_view" "'+view+'"\n'
ledger=case/'main.bean'
if split:
    (case/'sub').mkdir();target=case/'sub'/'entries.bean';target.write_text(base)
    ledger.write_text(settings+'include "sub/entries.bean"\n')
else:target=ledger;ledger.write_text(settings+base)
initial=case/'initial.bean';initial.write_text(settings+base)
control=case/'direct-control.bean';control.write_text(settings+base)
authored_before=case/'authored-before.bean';authored_before.write_text(base)
day='2024-02-02' if new_date else '2024-02-01'
source.write_text(day+',100.01,0.02\n')
fixed={p:p.read_bytes() for p in (module,importer,source,initial,authored_before)}
env=os.environ.copy()
for key in ('BEA_FILE','BEA_TOKEN','BEANCOUNT_LOAD_CACHE_FILENAME','BEA_ENGINE_PYTHON','BEA_ENGINE_DIR'):
    env.pop(key,None)
overrides=dict(BEA_CONFIG_DIR=str(case/'config'),XDG_CACHE_HOME=str(case/'cache'),BEA_NO_UPDATE_NOTIFIER='1')
env.update(overrides)
def run(tag,args,file=ledger,human_output=False):
    watched={p:p.read_bytes() for p in set((ledger,target,control))}
    argv=['bea',*([] if human_output else ['--json']),'--no-input','-f',str(file),*map(str,args)]
    started=time.monotonic()
    r=subprocess.run(argv,cwd=case,env=env,stdin=subprocess.DEVNULL,capture_output=True,text=True,timeout=25)
    assert all(p.read_bytes()==b for p,b in fixed.items())
    after={p:p.read_bytes() for p in watched}
    print(json.dumps(dict(tag=tag,args=argv,cwd=str(case),stdin_mode='DEVNULL',tty=False,env=overrides,
                         duration=time.monotonic()-started,exit=r.returncode,stdout=r.stdout,stderr=r.stderr,
                         input_sha256={p.name:hashlib.sha256(b).hexdigest() for p,b in fixed.items()},
                         before={str(p.relative_to(case)):hashlib.sha256(b).hexdigest() for p,b in watched.items()},
                         after={str(p.relative_to(case)):hashlib.sha256(b).hexdigest() for p,b in after.items()})),flush=True)
    if r.returncode:assert after==watched,'Inspect unexpected failed-write effects before retrying'
args=['import',source,'--config',importer,*(['--into','sub/entries.bean'] if split else []),*(['--duplicates','skip'] if skip else [])]
run('baseline',['check'])
run('direct-control',['add','balance','--date',day,'--account','Assets:Cash','--amount','100.01 ~ 0.02 USD'],control)
run('preview',args)
run('apply',[*args,'--apply'],human_output=human)
run('repeat',[*args,'--apply'])
run('check',['check'])
run('list',['list','balance','--limit','100'])
run('cash',['query',"SELECT sum(number) WHERE account='Assets:Cash'"])
run('control-check',['check'],control)
text=target.read_text()
if not split:
    assert text.startswith(settings);text=text[len(settings):]
(case/'authored-after.bean').write_text(text)

Cause: `cli/src/bea_engine/importing.py:517` initializes
`other_entries = _authored_directive_keys(snapshot)`. That helper (:2192)
correctly recovers declarations from every captured file, without plugins.
This preserves exact Balance repeat detection after completed
[1000](./done/1000.md), as the public exact repeats and six pilot cases prove.
However, the generic non-transaction branch (:781–785) checks only membership
of the **complete** `_directive_key`, including amount and tolerance. A new
100.01/0.02 assertion differs from the recorded 100.00/None key, so it remains
`new` and is included. There is no Balance-specific same-date/account/currency
conflict test analogous to the Price branch immediately above it.

The Balance-only post-pass (:830–851) checks whether a used pad would silently
fill it; these books have no pad, so it adds no block. `validate_append`
(:908) then sees the plugin's candidate view. Filtering removes the pair;
transforming changes both to the same 100.01/0.02 value. No differing duplicate
remains there, so preview/apply accept it. Ordinary pass-through keeps the pair
and correctly trips native validation. The final append (:1072 onward) uses
those accepted texts. This is a missing authored conflict decision, not a
request to generally bypass plugins or reject valid filtered imports.
