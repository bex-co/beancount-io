# Return update diffs that apply to the ledger

**Minor · cli · 40–55m · A1 · worker1.** An ordinary quoted memo containing
U+2028 or U+2029 makes `update transaction` and `update transactions` return
an invalid unified patch for a later entry. The diff invents an extra source
line and can start its context halfway through the earlier memo. JSON preview,
JSON apply and batch per-row diffs carry it; human preview prints the same
patch. `git apply --check` refuses it. The actual `--apply` edit is correct:
only the requested narration changes, and ledger validity and money stay intact.
A valid file without a final LF also yields an invalid patch: the shared
builder omits the no-newline marker. A leading UTF-8 byte-order mark is
also missing from the patch when its first line appears in the hunk. See
the additional confirmations below.
Agents need the displayed diff to describe the file they are reviewing.
Report mode; no product fix applied.

Environment: 2026-10-29, clean main and immutable runtime
`90de37eaad97a331568cf0b732ed364cff5acb79`, all **1,745** tracked CLI files
fingerprinted before/after and compared with freshly pulled main. Checkout
`bea 0.3.1`, frontend Python 3.13.3, existing managed Python 3.12.10 /
Beancount 3.2.3 / Beanquery 0.2.0, macOS 26.5.1 arm64,
Git 2.50.1 (Apple Git-155). Bounded fresh processes, closed stdin/no TTY,

```python

from pathlib import Path

case = Path(sys.argv[1]).absolute()
form, separator, bea, engine = sys.argv[2:6]
assert form in ('single', 'batch')
sep = {'u2028': '\u2028', 'u2029': '\u2029', 'space': ' ', 'lf': '\n'}[separator]
case.mkdir(parents=True, exist_ok=False)
book = case / 'main.bean'
before = '''option "operating_currency" "USD"
2024-01-01 open Assets:Cash USD
2024-01-01 open Expenses:Food USD
2024-01-01 open Equity:Opening USD
2024-01-01 * "Opening"
  Assets:Cash 100.00 USD
  Equity:Opening -100.00 USD
2024-01-02 ! "First'''+sep+'''continued"
  Assets:Cash -5.00 USD
  Expenses:Food 5.00 USD
2024-01-03 ! "Second"
  ref: "T2"
  Assets:Cash -7.00 USD
  Expenses:Food 7.00 USD
'''
book.write_bytes(before.encode())
after = before.replace('2024-01-03 ! "Second"', '2024-01-03 ! "Reviewed"')
env = os.environ.copy()
for key in ('BEA_FILE','BEA_TOKEN','BEANCOUNT_LOAD_CACHE_FILENAME','BEA_ENGINE_PYTHON','BEA_ENGINE_DIR'):
    env.pop(key, None)
overrides = dict(BEA_CONFIG_DIR=str(case/'config'), XDG_CACHE_HOME=str(case/'cache'), BEA_NO_UPDATE_NOTIFIER='1')
env.update(overrides)
result = dict(form=form, separator=separator, os=platform.platform(), python=platform.python_version(), before=before, expected_after=after, calls=[], native=[], git=[])
def save():
    (case/'results.json').write_text(json.dumps(result, indent=2, ensure_ascii=False))
def run(kind, name, argv, text=None, child_env=env):
    initial = hashlib.sha256(book.read_bytes()).hexdigest(); tick = time.monotonic()
    mode = dict(stdin=subprocess.DEVNULL) if text is None else dict(input=text)
    p = subprocess.run(list(map(str, argv)), cwd=case, env=child_env, capture_output=True, text=True, timeout=25, **mode)
    result[kind].append(dict(name=name, args=list(map(str,argv)), cwd=str(case), env=overrides, stdin_mode='DEVNULL' if text is None else 'piped patch', tty=False, exit=p.returncode, stdout=p.stdout, stderr=p.stderr, duration=round(time.monotonic()-tick,3), before_sha256=initial, after_sha256=hashlib.sha256(book.read_bytes()).hexdigest(), stdin=text))
    save(); return p
def cli(name, args, write=False, human=False):
    initial = book.read_bytes()
    p = run('calls', name, [bea, *([] if human else ['--json']), '--no-input', '--offline', '--file', book, *args])
    assert p.returncode == 0 and (write or book.read_bytes()==initial), (name,p.stderr)
    return p if human else json.loads(p.stdout)['data']
native_code = '''import json,sys
from decimal import Decimal as D
from beancount import loader
from beancount.core import data
loader.initialize(use_cache=False)
es,errors,_=loader.load_file(sys.argv[1]);balances={};rows=[]
for e in es:
 if isinstance(e,data.Transaction):
  rows.append([e.meta.get('ref'),e.narration])
  for p in e.postings:balances[p.account]=balances.get(p.account,D(0))+p.units.number
print(json.dumps(dict(valid=not errors,errors=[e.message for e in errors],rows=rows,balances={a:str(n) for a,n in balances.items()}),ensure_ascii=False))
'''
def graph(name):
    ne=env.copy(); ne.pop('PYTHONPATH',None); initial=book.read_bytes()
    p=run('native',name,[engine,'-B','-c',native_code,book],child_env=ne)
    assert p.returncode==0 and not p.stderr and book.read_bytes()==initial
    g=json.loads(p.stdout); assert g['valid'],g['errors']; return g
def physical(text):
    parts=text.split('\n'); return [line+'\n' for line in parts[:-1]]+([parts[-1]] if parts[-1] else [])
def portable(diff):
    lines=diff.split('\n'); assert lines[0].startswith('--- ') and lines[1].startswith('+++ ')
    lines[0]='--- a/main.bean'; lines[1]='+++ b/main.bean'; return '\n'.join(lines)
assert run('git','init',['git','-c','init.defaultBranch=main','init','--quiet']).returncode==0
assert run('calls','version',[bea,'--version']).returncode==0
g0=graph('before'); cli('initial-check',['check'])
if form=='batch':
    mapping=case/'mapping.json'; mapping.write_text(json.dumps([dict(id='T2',narration='Reviewed')]))
    args=['update','transactions','--from',mapping]
else:

preview=cli('preview',args); assert preview['written']==0
human=cli('human-preview',args,human=True); assert preview['diff'] in human.stdout
expected=''.join(difflib.unified_diff(physical(before),physical(after),fromfile=str(book),tofile=str(book)))
initial=book.read_bytes()
actual_check=run('git','returned-patch-check',['git','-c','core.autocrlf=false','apply','--check','-'],portable(preview['diff']))
control=run('git','physical-LF-patch-check',['git','-c','core.autocrlf=false','apply','--check','-'],portable(expected))
assert control.returncode==0 and book.read_bytes()==initial,control.stderr
applied=cli('apply',args+['--apply'],write=True)
assert applied['written']==1 and applied['diff']==preview['diff'] and book.read_bytes()==after.encode()
cli('final-check',['check']); g1=graph('after')
assert g0['balances']==g1['balances'] and g1['rows']==[g0['rows'][0],g0['rows'][1],['T2','Reviewed']]
result.update(actual_after=book.read_text(), preview=preview, applied=applied, expected_diff=expected, returned_patch_exit=actual_check.returncode, physical_LF_patch_exit=control.returncode, financial_data_preserved=True, human_preview_same_diff=True)
save()
print(json.dumps(dict(form=form,separator=separator,returned_patch_exit=actual_check.returncode,physical_LF_patch_exit=control.returncode,balances=g1['balances']),ensure_ascii=False))
```
