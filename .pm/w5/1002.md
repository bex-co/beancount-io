```python
from pathlib import Path
import hashlib, json, os, subprocess, time

case = Path('/abs/fallback-case')
method = 'LIFO'
partial = False
reorder_new = False
explicit_buy = False
only_existing = False
only_new = False
case.mkdir(parents=True, exist_ok=False)
ledger = case / 'main.bean'
fresh = case / 'control.bean'
base = f'''option "operating_currency" "USD"
2024-01-01 open Assets:Existing HOOL "{method}"
2024-01-01 open Assets:New HOOL "FIFO"
2024-01-01 open Assets:Cash USD
2024-01-01 open Income:Gains USD
2024-01-01 open Equity:Opening USD
2024-01-01 commodity HOOL
2024-01-01 * "Capital"
  Assets:Cash 1000 USD
  Equity:Opening -1000 USD
2024-01-02 * "Old holding"
  Assets:Existing 2 HOOL {{100 USD, "Old"}}
  Assets:Cash -200 USD
'''
ledger.write_text(base)
fresh.write_text(base)
rows = [
    {"date":"2024-02-01","narration":"Existing sale","postings":[
        {"account":"Assets:Existing","amount":"-1 HOOL {USD} @ 150 USD"},
        {"account":"Assets:Cash","amount":"150 USD"},
        {"account":"Income:Gains"}]},
    {"date":"2024-02-01","narration":"New sale","postings":[
        {"account":"Assets:New","amount":"-1 HOOL {USD} @ 150 USD"},
        {"account":"Assets:Cash","amount":"150 USD"},
        {"account":"Income:Gains"}]},
    {"date":"2024-02-01","narration":"Existing buy","postings":[
        {"account":"Assets:Existing","amount":"1 HOOL {200 USD, \"New\"}"},
        {"account":"Assets:Cash","amount":"-200 USD"}]},
    {"date":"2024-02-01","narration":"Inferred buy","postings":[
        {"account":"Assets:New","units":{"number":None,"currency":"HOOL"},
         "cost":{"number":"100","currency":"USD"}},
        {"account":"Assets:Cash","amount":"-100 USD"}]},
]
if explicit_buy:
    rows[3]['postings'][0]['units']['number'] = '1'
indices = [0,1,2,3]
correct = [0,3,1,2]
if only_existing:
    indices = correct = [0,2]
elif only_new:
    indices, correct = [1,3], [3,1]
elif reorder_new:
    indices = correct
batch = case / 'batch.json'
control = case / 'control.json'
batch.write_text(json.dumps([rows[i] for i in indices]))
control.write_text(json.dumps([rows[i] for i in correct]))
inputs = {p:p.read_bytes() for p in (batch, control)}
env = os.environ.copy()
for key in ('BEA_FILE', 'BEA_TOKEN', 'BEANCOUNT_LOAD_CACHE_FILENAME',
            'BEA_ENGINE_PYTHON', 'BEA_ENGINE_DIR'):
    env.pop(key, None)
env.update(BEA_CONFIG_DIR=str(case / 'config'), XDG_CACHE_HOME=str(case / 'cache'),
           BEA_NO_UPDATE_NOTIFIER='1')
def run(tag, args, file=ledger, json_mode=True):
    argv = ['bea', *(['--json'] if json_mode else []), '--no-input', '-f', str(file), *map(str, args)]
    before = file.read_bytes()
    started = time.monotonic()
    result = subprocess.run(argv, cwd=case, env=env, stdin=subprocess.DEVNULL,
                            capture_output=True, text=True, timeout=25)
    assert all(p.read_bytes() == data for p, data in inputs.items())
    print(json.dumps(dict(tag=tag, args=argv, exit=result.returncode,
                         stdout=result.stdout, stderr=result.stderr,
                         ledger_changed=file.read_bytes() != before,cwd=str(case),stdin_mode='DEVNULL',tty=False,
                         duration=time.monotonic()-started,env={k:env[k] for k in ('BEA_CONFIG_DIR','XDG_CACHE_HOME','BEA_NO_UPDATE_NOTIFIER')},
                         input_sha256={p.name:hashlib.sha256(p.read_bytes()).hexdigest() for p in inputs},
                         ledger_before=hashlib.sha256(before).hexdigest(),ledger_after=hashlib.sha256(file.read_bytes()).hexdigest())), flush=True)
report = ['report', 'balance-sheet', '--time', '2024-02', '--conversion', 'at_cost']
run('baseline', ['check'])
run('batch', ['add', 'transactions', '--from', batch, *(['--partial'] if partial else [])])
run('sale-lot', ['query', "SELECT number,cost_number,cost_date,cost_label WHERE account='Assets:Existing' AND narration='Existing sale'"])
run('report', report)
run('check', ['check'])
run('valid-order-control', ['add', 'transactions', '--from', control], file=fresh)
run('control-report', report, file=fresh)
run('control-check', ['check'], file=fresh)
```
